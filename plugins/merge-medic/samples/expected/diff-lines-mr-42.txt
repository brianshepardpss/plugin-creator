    1       +"""Late-fee calculation for overdue invoices."""
    2       +from decimal import ROUND_HALF_UP, Decimal
    3       +
    4       +GRACE_DAYS = 5
    5       +DAILY_RATE = Decimal("0.0005")  # 0.05% of the amount due per day late
    6       +FEE_CAP = Decimal("0.10")  # never charge more than 10% of the amount due
    7       +CENT = Decimal("0.01")
    8       +
    9       +
   10       +def late_fee(amount_due: Decimal, days_late: int) -> Decimal:
   11       +    """Return the late fee for an invoice that is `days_late` days overdue.
   12       +
   13       +    No fee inside the grace period. After it, the fee accrues daily and is
   14       +    capped at FEE_CAP of the amount due.
   15       +    """
   16       +    if days_late <= GRACE_DAYS:
   17       +        return Decimal("0.00")
   18       +    billable_days = days_late - GRACE_DAYS
   19       +    fee = amount_due * DAILY_RATE * billable_days
   20       +    capped = max(fee, amount_due * FEE_CAP)
   21       +    return capped.quantize(CENT, rounding=ROUND_HALF_UP)
